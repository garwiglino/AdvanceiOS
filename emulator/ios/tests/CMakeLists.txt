cmake_minimum_required(VERSION 3.19)
project(MGBAiOSCoreTests C)
set(CMAKE_C_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(READ "${ROOT}/ios/CoreSources.json" MANIFEST)
string(JSON COUNT LENGTH "${MANIFEST}")
math(EXPR LAST "${COUNT} - 1")
foreach(I RANGE ${LAST})
    string(JSON SOURCE GET "${MANIFEST}" ${I})
    list(APPEND SOURCES "${ROOT}/${SOURCE}")
endforeach()
add_executable(core_tests core_tests.c ${SOURCES} ../Bridge/MGBASession.c ../Bridge/AudioRing.c ../Bridge/MGBALauncher.c ../Bridge/MGBAInput.c ../Bridge/version.c)
target_include_directories(core_tests PRIVATE "${ROOT}/include" "${ROOT}/src" ../Bridge)
target_compile_definitions(core_tests PRIVATE M_CORE_GBA M_CORE_GB MINIMAL_CORE=2 DISABLE_THREADING USE_VFS_FILE BUILD_STATIC HAVE_STRDUP _GNU_SOURCE)
if(WIN32)
    target_link_libraries(core_tests PRIVATE ws2_32 shlwapi)
else()
    target_compile_definitions(core_tests PRIVATE HAVE_STRNDUP HAVE_LOCALTIME_R)
    target_link_libraries(core_tests PRIVATE m)
endif()
if(APPLE)
    target_compile_definitions(core_tests PRIVATE HAVE_STRLCPY HAVE_LOCALE HAVE_XLOCALE HAVE_SNPRINTF_L HAVE_STRTOF_L)
    target_link_libraries(core_tests PRIVATE "-framework CoreFoundation")
endif()
# Assertions are the test oracle, including when built with Release.
target_compile_options(core_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror=implicit-function-declaration -Wno-unused-parameter)
enable_testing()
add_test(NAME bridge_and_audio COMMAND core_tests)
